💻 Quick Start

<!-- 1. Embed the widget -->
<script 
  src="https://jovylle.com/widget/embed-inline.js"
  data-position="bottom-right"
  data-size="medium"
  async>
</script>

<!-- 2. Send notifications -->
<script>
window.JovylleInlineWidget.addNotification({
  type: 'success',
  title: 'Welcome!',
  message: 'Thanks for visiting!',
  persistent: true
});
</script>
